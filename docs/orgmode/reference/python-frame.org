* In-tree ~dseams.Frame~

~import dseams~ is the module built in this tree. ~dseams.Frame~ stores
a coordinate array. ~dseams.from_ase~ returns positions and a LAMMPS
cell, which ~set_frame~ accepts. The calls below are the ice Ih cell
from the Python recording: 128 waters, 512 neighbour entries, 256
six-rings, mean q6 0.625855, and 128 hexagonal labels.

#+begin_src python
>>> import numpy as np
>>> import dseams
>>> def ice():
...     xyz, edges, reading = [], [], False
...     handle = open("input/traj/ice1h.lammpstrj")
...     try:
...         while True:
...             line = handle.readline()
...             if line == "":
...                 break
...             if line.startswith("ITEM: BOX BOUNDS"):
...                 edges = []
...                 for _ in range(3):
...                     lo, hi = handle.readline().split()[:2]
...                     edges.append(float(hi) - float(lo))
...             elif line.startswith("ITEM: ATOMS"):
...                 reading = True
...             elif reading:
...                 parts = line.split()
...                 if len(parts) < 5:
...                     break
...                 xyz.append([float(v) for v in parts[2:5]])
...     finally:
...         handle.close()
...     return xyz, edges
>>> xyz, edges = ice()
>>> frame = dseams.Frame()
>>> frame.set_frame(np.asarray(xyz), np.asarray(edges))
>>> frame.count()
128
>>> offsets, cols = frame.neighbours(3.5)
>>> int(offsets[-1])
512
>>> sizes, sym, hist, frac = frame.rings(3.5, 6)
>>> int(hist[6])
256
>>> q4, q6 = frame.q4q6(3.5)
>>> round(float(q6.mean()), 6)
0.625855
>>> labels, counts = frame.chill_plus()
>>> int(counts[1])
128
>>> from ase import Atoms
>>> atoms = Atoms(symbols=["O"] * len(xyz), positions=xyz, cell=edges, pbc=True)
>>> pos, box = dseams.from_ase(atoms)
>>> pos.shape
(128, 3)
>>> frame.set_frame(pos, box)
>>> int(frame.chill_plus()[1][1])
128
#+end_src
